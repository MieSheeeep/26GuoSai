```python
@dataclass(frozen=True)
class HalfPlane:
    a: float
    b: float
    c: float
    def contains(self, point, tol=1e-9):
        return self.a * point[0] + self.b * point[1] <= self.c + tol

def bearing_halfplanes(station, bearing_deg, error_deg=1.0):
    x, y = station
    lo, hi = radians(bearing_deg-error_deg), radians(bearing_deg+error_deg)
    ulo, uhi = (cos(lo), sin(lo)), (cos(hi), sin(hi))
    return (HalfPlane(ulo[1], -ulo[0], ulo[1]*x-ulo[0]*y),
            HalfPlane(-uhi[1], uhi[0], -uhi[1]*x+uhi[0]*y))
```

```python
def localization_polygon(observations, tol=1e-9):
    planes = [p for obs in observations for p in bearing_halfplanes(*obs)]
    candidates = [intersection(p, q) for p, q in combinations(planes, 2)]
    feasible = [x for x in candidates if x is not None and all(p.contains(x, tol) for p in planes)]
    hull = convex_hull(feasible, tol)
    if len(hull) < 3:
        raise ValueError("constraints do not form a bounded polygon")
    return hull
```

```python
def polygon_diameter(vertices):
    n, j, best = len(vertices), 1, (-1.0, vertices[0], vertices[0])
    for i in range(n):
        ni = (i + 1) % n
        while abs(cross(vertices[i], vertices[ni], vertices[(j+1) % n])) > abs(cross(vertices[i], vertices[ni], vertices[j])) + 1e-12:
            j = (j + 1) % n
        for a, b in ((vertices[i], vertices[j]), (vertices[ni], vertices[j])):
            candidate = (squared_distance(a, b), a, b)
            if candidate[0] > best[0]:
                best = candidate
    return sqrt(best[0]), best[1], best[2]

def diameter_circle_covers(vertices, tol=1e-9):
    diameter, p, q = polygon_diameter(vertices)
    center = ((p[0] + q[0]) / 2, (p[1] + q[1]) / 2)
    return all(dist(center, v) <= diameter / 2 + tol for v in vertices), center, diameter / 2
```
